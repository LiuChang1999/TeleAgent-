cmake_minimum_required(VERSION 3.20)
project(TeleAgentAccountManager LANGUAGES CXX RC)
add_executable(TeleAgentAccounts WIN32 src/webui.cpp src/storage.cpp src/control.cpp src/installation.cpp src/tray.cpp src/desk.cpp src/polling.cpp src/embedded_loader.cpp src/settings.cpp app.rc)
set_property(SOURCE app.rc PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ui/index.html;${CMAKE_CURRENT_SOURCE_DIR}/app.manifest;${CMAKE_CURRENT_SOURCE_DIR}/assets/account-desk.ico;${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/x64/WebView2Loader.dll;${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/LICENSE.txt")
target_compile_features(TeleAgentAccounts PRIVATE cxx_std_20)
target_compile_definitions(TeleAgentAccounts PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(TeleAgentAccounts PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(TeleAgentAccounts SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2")
target_link_libraries(TeleAgentAccounts PRIVATE shell32 advapi32 ole32 uuid dwmapi comdlg32)
if(MINGW)
  set_property(SOURCE app.rc PROPERTY COMPILE_FLAGS "--use-temp-file")
  # Recent MinGW toolchains inject another manifest through their endfile spec.
  # Retain all normal startup files while using our explicit application manifest.
  execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpspecs OUTPUT_VARIABLE TA_GCC_SPECS)
  string(REGEX MATCH "\\*endfile:\n[^\n]*" TA_ENDFILE_SPEC "${TA_GCC_SPECS}")
  if(TA_ENDFILE_SPEC MATCHES "default-manifest.o")
    string(REPLACE "%{!shared:%:if-exists(default-manifest.o%s)}" "" TA_ENDFILE_SPEC "${TA_ENDFILE_SPEC}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/app-manifest.specs" "${TA_ENDFILE_SPEC}\n\n")
    target_link_options(TeleAgentAccounts PRIVATE "-specs=${CMAKE_CURRENT_BINARY_DIR}/app-manifest.specs")
  endif()
  target_compile_options(TeleAgentAccounts PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  target_link_options(TeleAgentAccounts PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
